• las variables y su tipo
• los operadores
• las funciones y sus parámetros
• las estructuras de control

<?php
function doble($i) {
 return $i*2;
}
$a = TRUE;            VARIABLE TIPO BOOLEAN
$b = "xyz";           VARIABLE TIPO STRING
$c = 'xyz';           VARIABLE TIPO STRING
$d = 12;              VARIABLE TIPO INTEGER
echo gettype($a);
echo gettype($b);
echo gettype($c);
echo gettype($d);
if (is_int($d)) {
 $d += 4;
}
if (is_string($a)) {
 echo "Cadena: $a";
}
$d = $a ? ++$d : $d*3;
$f = doble($d++);      VARIABLE TIPO INTEGER
$g = $f += 10;         VARIABLE TIPO INTEGER
echo $a, $b, $c, $d, $f , $g;
?>
